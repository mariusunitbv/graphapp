PROGRAM PDF
BEGIN
  U := N − {s}; V := {s}; W := ∅; #init
  FOR toți y ∈ N DO p(y) := -; #init
  t := 1; t1(s) := 1; t2(s) := ∞; #init
  FOR toți y ∈ U DO t1(y) := ∞; t2(y) := ∞; #init
  WHILE V ≠ ∅ DO
  BEGIN
    se selectează cel mai nou nod x introdus în V; #analyzing
    IF există arc(x, y) ∈ A și y ∈ U #visit
    THEN U := U − {y}; V := V ∪ {y}; p(y) := x; #visit
      t := t + 1; t1(y) := t #visit
    ELSE V := V − {x}; W := W ∪ {x}; #analyzed
      t := t + 1; t2(x) := t; #analyzed
   END;
END #end
